三、G7：正常写入与失败重试都保护另一原包

本次 R-G6-WRITE-OWNER-01/P2：useLessonOperation.run 写入恢复包前没有读当前归属；retryRecoveryWrite 的 write 分支只 validate 旧包形状，不核完整包身份。两个已加载页面共享 create/import 键；B 的合法原包结果未知时，A 正常开始或缓存写入失败后重试可覆盖 B。刷新 B 会读到 A 包，丢失 B 持久恢复身份。不是已证实正文丢失、重复服务端创建或数据库损坏。

pseudocode:
function writeOwnedPackage(adapter, operation, contextKey, isCurrent) {
  requireSameLiveSession(isCurrent);
  validateOperation(operation, contextKey);
  const existing = adapter.read(); // 读取失败/坏包不能当 empty
  if (existing !== null) {
    validateOperation(existing, contextKey);
    require(exactFrozenOperationKey(existing) === exactFrozenOperationKey(operation));
    // 同 context 但不同 submission/正文/metadata 的合法包也必须保护
  }
  requireSameLiveSession(isCurrent);
  adapter.prepare?.(operation); // 不在确认 foreign 之前改变旁路恢复身份

  adapter.write(operation);
  require(exactFrozenOperationKey(adapter.read()) === exactFrozenOperationKey(operation));
  requireSameLiveSession(isCurrent);
}

async beforeHTTP(operation) {

function validateNativePageImage(ref, root) {
